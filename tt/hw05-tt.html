selected_fields <- c(
  "Engineering",
  "Computer Science",
  "Business",
  "Health Professions"
)

major_comparison <- ai_salary_majors |>
  filter(broad_field_short %in% selected_fields) |>
  drop_na(
    median_starting_salary,
    ai_exposure_percentile
  )

major_comparison |>
  count(broad_field_short)
# A tibble: 4 × 2
  broad_field_short      n
  <chr>              <int>
1 Business              19
2 Computer Science       8
3 Engineering           38
4 Health Professions    18
Code
field_summary <- major_comparison |>
  group_by(broad_field_short) |>
  summarise(
    n_majors = n(),
    median_earnings = median(median_starting_salary),
    median_exposure = median(ai_exposure_percentile),
    .groups = "drop"
  ) |>
  arrange(desc(median_earnings))

field_summary

field_order <- field_summary |>
  arrange(median_earnings) |>
  pull(broad_field_short)

major_comparison <- major_comparison |>
  mutate(
    broad_field_short = factor(
      broad_field_short,
      levels = field_order
    )
  )
Code
plot1 <- ggplot(
  major_comparison,
  aes(x = median_starting_salary, y = broad_field_short)
) +
  geom_boxplot(
    orientation = "y",
    width = 0.5,
    fill = "#DDEFF0",
    color = "#087F8C",
    alpha = 0.65,
    outlier.shape = NA
  ) +
  geom_point(
    position = position_jitter(
      width = 0,
      height = 0.12,
      seed = 253
    ),
    color = "#087F8C",
    size = 2.4,
    alpha = 0.65
  ) +
  scale_x_continuous(
    limits = c(0, NA),
    expand = expansion(mult = c(0, 0.05))
  ) +
  labs(
    title = "Early-career earnings across fields",
    subtitle = paste(
      "Each point is one major;",
      "fields are ordered by their median earnings."
    ),
    x = "Earnings 1 year after graduation",
    y = NULL,
    caption = paste(
      "Author: Jason You",
      "\nSource: US Department of Education, AIOE Index, Tidytuesday"
    )
    
  ) +
  theme_minimal(
  )

library(patchwork)
combined_plot <- (
  plot1 +
    labs(
      title = "Early-career earnings",
      subtitle = NULL,
      caption = NULL,
      y = NULL
    )
) | (
  plot2 +
    labs(
      title = "AI exposure",
      subtitle = NULL,
      caption = NULL,
      y = NULL
    ) +
    theme(
      axis.text.y = element_blank(),
      axis.ticks.y = element_blank()
    )
)

combined_plot <- combined_plot +
  plot_layout(widths = c(1.3, 1)) +
  plot_annotation(
    title = "Earning and AI exposure across different fields",
    caption = paste(
      "Author: Jason You",
      "\nSource: US Department of Education, AIOE Index, TidyTuesday"
    ),
    theme = theme(
      plot.title = element_text(face = "bold", size = 18),
      plot.caption = element_text(hjust = 0),
      plot.background = element_rect(
