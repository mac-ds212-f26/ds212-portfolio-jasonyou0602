ggplot(
  fiction_deaths,
  aes(
    x = confidence,
    fill = if_else(undone, "Reversed", "Not reversed")
  )
) +
  geom_bar() +
  geom_text(
    stat = "count",
    aes(label = after_stat(count)),
    position = position_stack(vjust = 0.5),
    size = 3
  ) +
  facet_wrap(~ franchise_slug, ncol = 4) +
  labs(
    title = "Recorded deaths by franchise, evidence, and reversal",
    subtitle = "40 of 307 recorded deaths were later reversed (13.0%).",
    x = "Evidence type",
    y = "Number of recorded deaths",
    fill = "Death outcome",
    caption = paste0(
      "Shown = directly depicted; stated = reported by a character or narrator.\n",
      "Author: Jason You | Source: Cited Feats / TidyTuesday, 2026-11-10"
    )
  ) +
  theme(
    plot.caption = element_text(hjust = 0)
